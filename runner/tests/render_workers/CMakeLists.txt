cmake_minimum_required(VERSION 3.20)
project(SnesRenderWorkersTests LANGUAGES C)
include(CTest)
include("${CMAKE_CURRENT_LIST_DIR}/../../render_workers.cmake")
add_executable(render_workers_test render_workers_test.c)
snesrecomp_target_render_workers(render_workers_test)
if(NOT MSVC)
    target_compile_options(render_workers_test PRIVATE -Wall -Wextra -Werror)
    target_compile_options(snesrecomp_render_workers PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME render_workers COMMAND render_workers_test)
set_tests_properties(render_workers PROPERTIES TIMEOUT 30)
