project(presenter_edges C)
find_package(SDL3 CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
add_executable(presenter_edges presenter_edges.c ../../../third_party/gl_core/gl_core_3_1.c)

target_link_libraries(presenter_edges PRIVATE SDL3::SDL3 OpenGL::GL)
enable_testing()
add_test(NAME presenter_edges COMMAND presenter_edges)
