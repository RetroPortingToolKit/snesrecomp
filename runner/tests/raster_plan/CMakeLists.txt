cmake_minimum_required(VERSION 3.20)
project(snes_raster_plan_tests C)
enable_testing()
add_executable(snes_raster_plan_test raster_plan_test.c)
target_include_directories(snes_raster_plan_test PRIVATE ../../src)
target_compile_features(snes_raster_plan_test PRIVATE c_std_11)
if(MSVC)
  target_compile_options(snes_raster_plan_test PRIVATE /UNDEBUG)
else()
  target_compile_options(snes_raster_plan_test PRIVATE -UNDEBUG)
endif()
add_test(NAME snes_raster_plan COMMAND snes_raster_plan_test)
