cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(snes_raster_snapshot_tests C)
    enable_testing()
    option(SNES_RASTER_TEST_TRACK_WRITES "Exercise optional PPU capture revisions" ON)
    add_compile_definitions(SNESRECOMP_PPU_CAPTURE_REVISION=$<BOOL:${SNES_RASTER_TEST_TRACK_WRITES}>)
endif()
add_executable(snes_raster_snapshot_test raster_snapshot_test.c)
target_include_directories(snes_raster_snapshot_test PRIVATE ../../src)
target_compile_features(snes_raster_snapshot_test PRIVATE c_std_11)
if(MSVC)
    target_compile_options(snes_raster_snapshot_test PRIVATE /UNDEBUG)
else()
    target_compile_options(snes_raster_snapshot_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()
add_test(NAME snes_raster_snapshot COMMAND snes_raster_snapshot_test)
add_executable(snes_ppu_capture_revision_test ppu_revision_test.c ../../src/snes/ppu.c)
target_include_directories(snes_ppu_capture_revision_test PRIVATE ../../src)
target_compile_features(snes_ppu_capture_revision_test PRIVATE c_std_11)
target_compile_definitions(snes_ppu_capture_revision_test PRIVATE
    SNESRECOMP_TRACE=0 SNESRECOMP_ENABLE_MODS=0)
if(MSVC)
    target_compile_options(snes_ppu_capture_revision_test PRIVATE /UNDEBUG)
else()
    target_compile_options(snes_ppu_capture_revision_test PRIVATE -UNDEBUG)
endif()
if(NOT WIN32)
    target_link_libraries(snes_ppu_capture_revision_test PRIVATE m)
endif()
add_test(NAME snes_ppu_capture_revision COMMAND snes_ppu_capture_revision_test)
add_executable(snes_raster_snapshot_benchmark raster_snapshot_benchmark.c)
target_include_directories(snes_raster_snapshot_benchmark PRIVATE ../../src)
target_compile_features(snes_raster_snapshot_benchmark PRIVATE c_std_11)
