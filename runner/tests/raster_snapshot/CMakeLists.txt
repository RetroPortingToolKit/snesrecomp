cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(snes_raster_snapshot_tests C)
    enable_testing()
endif()
add_executable(snes_raster_snapshot_test raster_snapshot_test.c)
target_include_directories(snes_raster_snapshot_test PRIVATE ../../src)
target_compile_features(snes_raster_snapshot_test PRIVATE c_std_11)
if(MSVC)
    target_compile_options(snes_raster_snapshot_test PRIVATE /UNDEBUG)
else()
    target_compile_options(snes_raster_snapshot_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()
add_test(NAME snes_raster_snapshot COMMAND snes_raster_snapshot_test)
add_executable(snes_raster_snapshot_benchmark raster_snapshot_benchmark.c)
target_include_directories(snes_raster_snapshot_benchmark PRIVATE ../../src)
target_compile_features(snes_raster_snapshot_benchmark PRIVATE c_std_11)
