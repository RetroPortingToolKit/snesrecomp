cmake_minimum_required(VERSION 3.16)
project(snes_input_schema_test C)
set(CMAKE_C_STANDARD 11)
get_filename_component(SNES_ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(RNET_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RNET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RNET_ENABLE_ICE OFF CACHE BOOL "" FORCE)
add_subdirectory("${SNES_ROOT}/lib/recomp-net" recomp-net)
add_executable(snes_input_schema_test input_schema_test.c)
target_include_directories(snes_input_schema_test PRIVATE "${SNES_ROOT}/runner/src")
target_link_libraries(snes_input_schema_test PRIVATE recomp_net)
if(MSVC)
    target_compile_options(snes_input_schema_test PRIVATE /UNDEBUG)
else()
    target_compile_options(snes_input_schema_test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME snes_input_schema COMMAND snes_input_schema_test)
